📜 Mã Google Apps Script (Tự động tạo 3 Sheet lưu dữ liệu)
Mở file Google Sheets trên Drive → Tiện ích mở rộng (Extensions) → Apps Script → Dán mã này và bấm Triển khai (Deploy) → Ứng dụng Web: 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);

  // 1. Sheet Nhật ký Ca (SHIFTS_DB)
  var shShift = ss.getSheetByName("1_NHAT_KY_CA") || ss.insertSheet("1_NHAT_KY_CA");
  if (shShift.getLastRow() === 0) {
    shShift.appendRow(["Ma_Ca", "Ngay", "Ma_Tro_Choi", "Ten_Tro_Choi", "Nhan_Vien", "Gio_Mo_Cua", "Gio_Dong_Cua", "Gio_VH_Thuc_Te_Phut", "Tong_Phut_Dung", "Tong_So_Luot", "Tong_Khach", "Thoi_Gian_Dong_Bo"]);
  }
  (payload.shifts || []).forEach(function(s) {
    shShift.appendRow([s.shiftId, s.date, s.gameId, s.gameName, s.staff, s.openTime, s.closeTime, s.netMin, s.totalDowntimeMin, s.totalCycles, s.totalPax, new Date()]);
  });

  // 2. Sheet Ngừng vận hành (DOWNTIMES_DB)
  var shDt = ss.getSheetByName("2_NGUNG_VAN_HANH") || ss.insertSheet("2_NGUNG_VAN_HANH");
  if (shDt.getLastRow() === 0) {
    shDt.appendRow(["Ma_Dung", "Ngay", "Ten_Tro_Choi", "Nhan_Vien", "Nguyen_Nhan", "Bat_Dau", "Ket_Thuc", "So_Phut_Dung", "Ghi_Chu"]);
  }
  (payload.downtimes || []).forEach(function(d) {
    shDt.appendRow([d.id, d.date, d.gameName, d.staff, d.reason, d.start, d.end, d.durationMin, d.note]);
  });

  // 3. Sheet Lượt chơi & Line/Rạp (CYCLES_DB)
  var shCy = ss.getSheetByName("3_LUOT_CHOI_CHI_TIET") || ss.insertSheet("3_LUOT_CHOI_CHI_TIET");
  if (shCy.getLastRow() === 0) {
    shCy.appendRow(["Ma_Luot", "Ngay", "Ten_Tro_Choi", "Khu_Vuc_Line_Rap", "Luot_So", "Phan_Loai", "Bat_Dau", "Ket_Thuc", "Chu_Ky_Giay", "So_Khach"]);
  }
  (payload.cycles || []).forEach(function(c) {
    shCy.appendRow([c.id, c.date, c.gameName, c.subArea, c.cycleNo, c.subType, c.start, c.end, c.durationSec, c.pax]);
  });

  return ContentService.createTextOutput(JSON.stringify({ status: "ok", syncedAt: new Date() }))
    .setMimeType(ContentService.MimeType.JSON);
}
